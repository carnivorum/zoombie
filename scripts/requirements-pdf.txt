# Python dependencies for the zoombie PDF -> Markdown toolchain.
#
# Installed by the zoombie installer with `pip install --user` into the SAME
# interpreter the toolchain already uses for yt-dlp, so there is one Python to
# maintain, no elevation is needed, and nothing is written into a system
# location.
#
#   pymupdf4llm  Markdown-aware extraction (headings, lists, tables) on top of PyMuPDF
#   pymupdf      PyMuPDF itself: PDF parsing, and page rasterization for OCR
#   pytesseract  thin wrapper over the Tesseract OCR engine (optional fallback)
#   Pillow       image objects handed to pytesseract

pymupdf4llm>=0.0.17
pymupdf>=1.24.0
pytesseract>=0.3.10
Pillow>=10.0.0
